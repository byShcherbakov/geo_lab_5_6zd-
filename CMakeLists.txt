cmake_minimum_required(VERSION 3.16)
project(OpenGL_Lab5_Variant2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Gui)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Gui)
if(QT_VERSION_MAJOR EQUAL 6)
    find_package(Qt6 REQUIRED COMPONENTS OpenGL OpenGLWidgets)
endif()
add_executable(OpenGL_Lab5_Variant2 main.cpp)
target_link_libraries(OpenGL_Lab5_Variant2 PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Gui)
if(QT_VERSION_MAJOR EQUAL 6)
    target_link_libraries(OpenGL_Lab5_Variant2 PRIVATE Qt6::OpenGL Qt6::OpenGLWidgets)
endif()
if(MSVC)
    target_compile_options(OpenGL_Lab5_Variant2 PRIVATE /utf-8 /W4)
else()
    target_compile_options(OpenGL_Lab5_Variant2 PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(OpenGL_Lab5_Variant2 PROPERTIES MACOSX_BUNDLE TRUE)
